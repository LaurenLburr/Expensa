    SELECT p.PayeeId,
           p.PayeeName,
           p.SortIndex,
           COALESCE(t.TagName, 'None') AS TagName,
           COALESCE(t.SortIndex, 9999) AS TagSortIndex

           LEFT JOIN
           Tag t ON t.TagId = pt.TagId;
